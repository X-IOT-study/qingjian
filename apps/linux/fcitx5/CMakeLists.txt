# //! 第一阶段只链接 Fcitx5 默认面板与 JSON IPC。
cmake_minimum_required(VERSION 3.16)
project(qingjian-fcitx5 VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Fcitx5Core 5.0.14 REQUIRED)
find_package(nlohmann_json 3.2 REQUIRED)
# 各 API 首次出现的版本；低于它的系统走 src/compat/ 里的实现，调用处不写 #if。
# 改兼容范围时只动这张表，门槛与降级行为见 docs/notes/linux-fcitx5-compat.md。
function(qingjian_require_fcitx5 feature version)
    if(Fcitx5Core_VERSION VERSION_GREATER_EQUAL ${version})
        set(enabled 1)
    else()
        set(enabled 0)
    endif()
    set(QINGJIAN_FCITX_HAS_${feature} ${enabled} PARENT_SCOPE)
    set(QINGJIAN_FCITX_DEFINITIONS "${QINGJIAN_FCITX_DEFINITIONS};QINGJIAN_FCITX_HAS_${feature}=${enabled}" PARENT_SCOPE)
endfunction()
qingjian_require_fcitx5(DISABLE 5.0.20)
qingjian_require_fcitx5(VIRTUAL_KEYBOARD 5.1.0)
qingjian_require_fcitx5(CANDIDATE_COMMENT 5.1.10)
qingjian_require_fcitx5(CUSTOM_IM_INFO 5.1.11)
qingjian_require_fcitx5(ADDON_FACTORY_V2 5.1.12)
qingjian_require_fcitx5(OVERLAY_MESSAGE 5.1.22)
include(CTest)
set(QINGJIAN_SOURCES src/qingjian.cpp src/ipc/connection.cpp src/key/mapping.cpp src/settings/command.cpp
    src/compat/candidate.cpp src/compat/input_method_info.cpp src/compat/virtual_keyboard.cpp)
add_library(qingjian MODULE src/factory.cpp ${QINGJIAN_SOURCES})
target_include_directories(qingjian PRIVATE src)
target_compile_definitions(qingjian PRIVATE ${QINGJIAN_FCITX_DEFINITIONS})
target_link_libraries(qingjian PRIVATE Fcitx5::Core nlohmann_json::nlohmann_json ${CMAKE_DL_LIBS})
set_target_properties(qingjian PROPERTIES PREFIX "")
install(TARGETS qingjian LIBRARY DESTINATION lib/fcitx5)
install(FILES data/addon/qingjian.conf DESTINATION share/fcitx5/addon)
install(FILES data/inputmethod/qingjian.conf DESTINATION share/fcitx5/inputmethod)
if(BUILD_TESTING)
